// ==================== tools/check_real_seg_http.py ====================
"""通过 HTTP 接口用真实 U-Net 权重跑一次分割，验证 PyTorch 插件在服务内可用。
用法：python tools/check_real_seg_http.py --base http://127.0.0.1:5000
"""
from __future__ import annotations
import argparse
import json
import sys
import urllib.error
import urllib.request
from http.cookiejar import CookieJar
from pathlib import Path
ROOT = Path(__file__).resolve().parent.parent
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))
ARCH_WEIGHTS = {
    "unet": "Unet_1000_best.pth",
    "resunet": "resunet_1000_best.pth",
    "att_resunet": "Atteresunet_1000_best.pth",
    "unetv2": "UNetV2_800_best.pth",
}
class Client:
    def __init__(self, base: str) -> None:
        self.base = base.rstrip("/")
        self.opener = urllib.request.build_opener(urllib.request.HTTPCookieProcessor(CookieJar()))
        self.token = ""
    def call(self, method: str, path: str, body: dict | None = None) -> dict:
        data = json.dumps(body).encode() if body is not None else None
        request = urllib.request.Request(self.base + path, data=data, method=method)
        request.add_header("Content-Type", "application/json")
        if self.token:
            request.add_header("Authorization", "Bearer " + self.token)
        try:
            with self.opener.open(request, timeout=300) as response:
                return json.loads(response.read())
        except urllib.error.HTTPError as exc:
            text = exc.read().decode("utf-8", "ignore")
            try:
                return json.loads(text)
            except json.JSONDecodeError:
                return {"code": exc.code, "message": text[:200]}
def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--base", default="http://127.0.0.1:5000")
    parser.add_argument("--arch", default=None, help="只测某个结构，默认全测")
    args = parser.parse_args()
    client = Client(args.base)
    login = client.call("POST", "/api/v1/auth/login",
                        {"username": "admin", "password": "admin123"})
    if login.get("code") != 0:
        print("登录失败:", login.get("message"))
        return 1
    client.token = login["data"]["token"]
    found: dict[str, dict] = {}
    for page in (1, 2, 3):
        listing = client.call("GET", f"/api/v1/models?task_type=segmentation&page_size=100&page={page}")
        for model in listing.get("data") or []:
            if model["variant"] == "1000" or model["name"].endswith("_best"):
                found.setdefault(model["arch"], model)
            found.setdefault(model["arch"], model)
    print("已登记的分割结构:", sorted(set(found)))
    targets = [args.arch] if args.arch else list(ARCH_WEIGHTS)
    failures = 0
    for arch in targets:
        model = found.get(arch)
        if model is None:
            print(f"[SKIP] {arch}: 模型仓库中没有该结构")
            continue
        config = client.call("POST", "/api/v1/pipeline/config",
                             {"mode": "segmentation", "segmenter_key": arch,
                              "segmenter_model_id": model["id"], "threshold": 0.30})
        errors = ((config.get("data") or {}).get("load_errors")) or []
        result = client.call("POST", "/api/v1/pipeline/process", {})
        road = ((result.get("data") or {}).get("road")) or {}
        ok = not errors and road.get("mask_area_ratio") is not None
        if not ok:
            failures += 1
        print(f"[{'OK ' if ok else 'FAIL'}] {arch:<12} 权重={model['name']:<22} "
              f"掩膜占比={road.get('mask_area_ratio')}  分割耗时={road.get('infer_ms')} ms  "
              f"左边界={((road.get('left') or {}).get('min') or {}).get('depth_m')} m  "
              f"走廊={road.get('corridor_width_m')} m")
        if errors:
            print("        加载错误:", errors)
        if road.get("arch"):
            print(f"        实际使用结构: {road.get('arch')}")
    client.call("POST", "/api/v1/pipeline/config",
                {"mode": "both", "segmenter_key": "classical_road", "segmenter_model_id": None})
    print("\n" + ("全部结构通过服务端验证" if not failures else f"{failures} 个结构未通过"))
    return 1 if failures else 0
if __name__ == "__main__":
    raise SystemExit(main())

PLAN = [
    ("unet", "Unet_1000_best.pth", "UNet"),
    ("resunet", "resunet_1000_best.pth", "ResUNet"),
    ("att_resunet", "Atteresunet_1000_best.pth", "Attention-ResUNet"),
    ("unetv2", "UNetV2_800_best.pth", "UNetV2 (PVTv2)"),

def metrics(pred_mask: np.ndarray, gt_mask: np.ndarray) -> dict:
    p = (pred_mask > 0).astype(np.uint8).ravel()
    g = (gt_mask > 0).astype(np.uint8).ravel()
    tp = int(np.logical_and(p == 1, g == 1).sum())
    fp = int(np.logical_and(p == 1, g == 0).sum())
    fn = int(np.logical_and(p == 0, g == 1).sum())
    tn = int(np.logical_and(p == 0, g == 0).sum())
    eps = 1e-7
    return {
        "IoU": round(tp / (tp + fp + fn + eps), 4),
        "Dice": round(2 * tp / (2 * tp + fp + fn + eps), 4),
        "Precision": round(tp / (tp + fp + eps), 4),
        "Recall": round(tp / (tp + fn + eps), 4),
        "Accuracy": round((tp + tn) / (tp + tn + fp + fn + eps), 4),
    }
def load_gt(mask_path: Path) -> np.ndarray | None:
    raw = cv2.imread(str(mask_path), cv2.IMREAD_UNCHANGED)
    if raw is None:
        return None
    if raw.ndim == 3:
        raw = raw[:, :, 0]
    return (raw > 0).astype(np.uint8)
def main() -> int:
    data_root = Path(sys.argv[1]) if len(sys.argv) > 1 else DEFAULT_DATA
    weights_dir = Path(sys.argv[2]) if len(sys.argv) > 2 else data_root / "models"
    setup_logging()
    registry.load_builtin()
    out_dir = settings.storage_dir("results") / "seg_compare"
    out_dir.mkdir(parents=True, exist_ok=True)
    test_dir = data_root / "dataset" / "test" / "images"
    masks_dir = data_root / "dataset" / "test" / "masks"
    image_path = test_dir / "Campus_road_196.jpg"
    if not image_path.exists():
        candidates = sorted(test_dir.glob("*.jpg")) if test_dir.exists() else []
        if not candidates:
            print(f"找不到测试图像，请检查路径：{test_dir}")
            return 1
        image_path = candidates[0]
    image = media.read_image(image_path)
    if image is None:
        print(f"图像读取失败：{image_path}")
        return 1
    gt = load_gt(masks_dir / (image_path.stem + ".png"))
    print(f"测试图像: {image_path.name}  {image.shape[1]}x{image.shape[0]}")
    if gt is not None:
        print(f"真值掩膜前景占比: {gt.mean():.4f}  shape={gt.shape}")
    panels = [image.copy()]
    labels = ["原图"]
    rows = []
    print("=" * 92)
    for arch, weight_name, label in PLAN:
        weight = weights_dir / weight_name
        plugin = registry.create("segmentation", arch)
        plugin.load(str(weight) if weight.exists() else None)
        pred = plugin.predict(image, threshold=0.30)
        mask = pred["mask"]
        ratio = float(mask.mean())
        row = {"模型": label, "掩膜占比": round(ratio, 4), "阈值语义": pred.get("output_activation")}
        if gt is not None:
            row.update(metrics(mask, gt))
        rows.append(row)
        vis = media.draw_mask(image.copy(), mask, alpha=0.35)
        vis = media.draw_panel(vis, [f"掩膜占比 {ratio:.3f}"], "top-right", label)
        panels.append(vis)
        labels.append(label)
        print("  ".join(f"{k}={v}" for k, v in row.items()))
    h, w = image.shape[:2]
    scale = 620 / w
    tw, th = int(w * scale), int(h * scale)
    tiles = [cv2.resize(p, (tw, th)) for p in panels]
    while len(tiles) < 6:
        tiles.append(np.zeros((th, tw, 3), dtype=np.uint8))
    grid = np.vstack([np.hstack(tiles[0:3]), np.hstack(tiles[3:6])])
    for index, label in enumerate(labels):
        x = (index % 3) * tw + 12
        y = (index // 3) * th + 12
        media.draw_text_items(grid, [{"text": label, "xy": (x, y), "color": (255, 255, 255),
                                      "size": 22, "bg": (40, 40, 40), "padding": 6}])
    out_path = out_dir / "seg_model_comparison.jpg"
    media.write_image(out_path, grid, quality=92)
    print("=" * 92)
    print("对比图已保存:", out_path)
    header = ["模型", "掩膜占比"] + (["IoU", "Dice", "Precision", "Recall", "Accuracy"]
                                    if gt is not None else [])
    print("\n| " + " | ".join(header) + " |")
    print("|" + "---|" * len(header))
    for row in rows:
        print("| " + " | ".join(str(row.get(k, "-")) for k in header) + " |")
    return 0
if __name__ == "__main__":
    raise SystemExit(main())

PLAN = [
    ("unet", "Unet_1000_best.pth"),
    ("resunet", "resunet_1000_best.pth"),
    ("att_resunet", "Atteresunet_1000_best.pth"),
    ("unetv2", "UNetV2_800_best.pth"),

def try_load(arch: str, weight: Path | None, size=(512, 1024)) -> dict:
    plugin = registry.create("segmentation", arch)
    t0 = time.perf_counter()
    try:
        plugin.load(str(weight) if weight else None)
    except Exception as exc:  # noqa: BLE001
        return {"arch": arch, "ok": False, "stage": "load", "error": str(exc)[:220]}
    load_ms = (time.perf_counter() - t0) * 1000
    image = (np.random.rand(size[0], size[1], 3) * 255).astype("uint8")
    t1 = time.perf_counter()
    try:
        pred = plugin.predict(image, threshold=0.3)
    except Exception as exc:  # noqa: BLE001
        return {"arch": arch, "ok": False, "stage": "predict", "error": str(exc)[:220]}
    infer_ms = (time.perf_counter() - t1) * 1000
    mask = pred.get("mask")
    return {
        "arch": arch, "ok": True, "load_ms": round(load_ms, 1), "infer_ms": round(infer_ms, 1),
        "device": plugin.device,
        "mask_shape": list(mask.shape) if mask is not None else None,
        "mask_ratio": round(float(mask.mean()), 4) if mask is not None else None,
        "activation": pred.get("output_activation"),
        "params_m": round(sum(p.numel() for p in plugin._model.parameters()) / 1e6, 2),  # noqa: SLF001
        "weight": weight.name if weight else None,
    }
def main() -> int:
    weights_dir = Path(sys.argv[1]) if len(sys.argv) > 1 else DEFAULT_WEIGHTS
    setup_logging()
    registry.load_builtin()
    print(f"PyTorch {torch.__version__}  CUDA={torch.cuda.is_available()}")
    print(f"权重目录: {weights_dir}  (存在={weights_dir.exists()})")
    print("=" * 92)
    failures = 0
    for arch, filename in PLAN:
        weight = weights_dir / filename
        if not weight.exists():
            weight = None
        result = try_load(arch, weight)
        if result["ok"]:
            print(f"[OK  ] {arch:<12} 权重={result['weight']}  参数={result['params_m']}M  "
                  f"加载={result['load_ms']}ms  推理={result['infer_ms']}ms  "
                  f"掩膜占比={result['mask_ratio']}  阈值语义={result['activation']}")
        else:
            failures += 1
            print(f"[FAIL] {arch:<12} 阶段={result['stage']}  {result['error']}")
    print("=" * 92)
    print("全部通过" if not failures else f"{failures} 个结构未通过")
    return 1 if failures else 0

    raise SystemExit(main())

"""验证真实 YOLO 权重能否加载并在真实道路图像上推理。
用法：

    python tools/check_yolo.py --all            # 遍历 20 个配置
    python tools/check_yolo.py --arch yolo12m   # 指定单一结构
"""
from __future__ import annotations
import argparse
import sys
import time
from pathlib import Path

if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))

DATASET = Path(r"E:\2Project\2026dachuang\Yolo_dis_v3源代码\datasets\images\val")
def pick_images(limit: int = 3) -> list[Path]:
    if DATASET.exists():
        files = sorted(p for p in DATASET.rglob("*") if p.suffix.lower() in (".jpg", ".jpeg", ".png"))
        if files:
            step = max(1, len(files) // limit)
            return files[::step][:limit]
    out = settings.storage_dir("results") / "check_algorithms" / "scene_rgb.jpg"
    return [out] if out.exists() else []
def main() -> int:
    parser = argparse.ArgumentParser(description="验证 YOLO 权重加载与推理")
    parser.add_argument("--arch", default=None, help="只测一个结构，如 yolo12m")
    parser.add_argument("--all", action="store_true", help="遍历全部 20 个配置")
    parser.add_argument("--mode", default="trained",
                        choices=["trained", "pretrained", "dual"],
                        help="权重模式（默认用自训练权重，类别与平台一致）")
    args = parser.parse_args()
    setup_logging()
    registry.load_builtin()
    images = pick_images(3)
    if not images:
        print("找不到可用于推理的测试图像。")
        return 1
    print(f"测试图像 {len(images)} 张，来自 {images[0].parent}")
    print(f"权重模式: {args.mode}")
    print("权重搜索目录：")
    for folder in settings.weight_search_dirs("detection"):
        count = len(list(folder.glob('*.pt')))
        print(f"  {folder}  ({count} 个 .pt)")
    configs = YoloDetectorAdapter.available_configs()
    if args.arch:
        configs = [c for c in configs if c["key"] == args.arch]
    elif not args.all:
        picks = {"yolov8n", "yolov9t", "yolov10m", "yolo11s", "yolo12m"}
        configs = [c for c in configs if c["key"] in picks]
    print(f"\n将测试 {len(configs)} 个配置\n" + "=" * 96)
    ok_count, fail_count = 0, 0
    total_classes, translated, rows = set(), set(), []
    for cfg in configs:
        version, size = cfg["version"].lstrip("v"), cfg["size"]
        key = cfg["key"]
        try:
            plugin = registry.create("detection", "yolo", version=version, size=size,
                                     weight_mode=args.mode)
            t0 = time.perf_counter()
            plugin.load(None)
            load_ms = (time.perf_counter() - t0) * 1000
            image = media.read_image(images[0])
            t1 = time.perf_counter()
            pred = plugin.predict(image, conf=0.25, iou=0.45, imgsz=640)
            infer_ms = (time.perf_counter() - t1) * 1000
            desc = plugin.describe()
            names = set(str(n) for n in (desc.get("class_names") or []))
            total_classes |= names
            sources = list((desc.get("sources") or {}).keys())
            labels = [str(x) for x in (pred.get("labels") or [])]
            translated |= set(labels)
            rows.append({"配置": key, "权重来源": "+".join(sources) or "-",
                         "类别数": len(names), "检出": len(pred.get("boxes") or []),
                         "加载(ms)": round(load_ms), "推理(ms)": round(infer_ms)})
            detail = ("  检出类别=" + ",".join(sorted(set(labels)))) if labels else ""
            print(f"[OK ] {key:<10} 权重={list((desc.get('sources') or {}).values())}  "
                  f"类别 {len(names)}  检出 {len(pred.get('boxes') or [])}  "
                  f"加载 {load_ms:.0f}ms 推理 {infer_ms:.0f}ms{detail}")
            ok_count += 1
        except Exception as exc:  # noqa: BLE001
            print(f"[ERR] {key:<10} {str(exc)[:150]}")
            fail_count += 1
    print("=" * 96)
    print(f"通过 {ok_count} 个，失败 {fail_count} 个")
    if rows:
        header = list(rows[0].keys())
        print("\n| " + " | ".join(header) + " |")
        print("|" + "---|" * len(header))
        for row in rows:
            print("| " + " | ".join(str(row[h]) for h in header) + " |")
    if total_classes:
        platform_classes = set(settings.detection_class_names())
        print(f"\n平台 11 类（中文）：{sorted(platform_classes)}")
        print(f"权重内置类别名（英文）：{sorted(total_classes)}")
        print(f"经平台别名映射后的检出类别：{sorted(translated) or '（本次抽检未检出目标）'}")
        unmapped = {t for t in translated if t not in platform_classes}
        if unmapped:
            print(f"[警告] 未被映射到平台类别体系的检出：{sorted(unmapped)}")
        else:
            print("[OK ] 所有权重类别名都能映射到平台的中文类别体系")
    return 1 if fail_count else 0
if __name__ == "__main__":
    raise SystemExit(main())

"""维护工具：清理验收/冒烟测试产生的演示实验，保留历史迁移实验。
用法：
    python tools/cleanup_demo.py            # 预览将删除的内容
    python tools/cleanup_demo.py --apply    # 实际删除
"""
from __future__ import annotations
import argparse
import sys
from pathlib import Path
ROOT = Path(__file__).resolve().parent.parent
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))
from core import db  # noqa: E402
from core.logging_setup import setup_logging  # noqa: E402
PATTERNS = ("端到端验收实验%", "实时实验%", "冒烟测试实验%", "默认实验%")
def main() -> int:
    parser = argparse.ArgumentParser(description="清理演示实验")
    parser.add_argument("--apply", action="store_true", help="实际执行删除")
    args = parser.parse_args()
    setup_logging()
    db.init_db()
    where = " OR ".join(["name LIKE ?"] * len(PATTERNS))
    rows = db.query_all(f"SELECT id, name, task_type, created_at FROM experiment "
                        f"WHERE is_deleted=0 AND ({where}) ORDER BY created_at",
                        PATTERNS)
    if not rows:
        print("没有需要清理的演示实验。")
        return 0
    print(f"发现 {len(rows)} 个演示实验：")
    total_records = 0
    for row in rows:
        count = int(db.query_scalar("SELECT COUNT(*) FROM ranging_result WHERE experiment_id=?",
                                     (row["id"],), 0) or 0)
        total_records += count
        print(f"  {row['id']}  {row['name']}  ({row['task_type']}, {count} 条测距记录)")
    reports = db.query_all(
        "SELECT id FROM report WHERE experiment_id IN "
        f"(SELECT id FROM experiment WHERE is_deleted=0 AND ({where}))", PATTERNS)
    print(f"关联报告 {len(reports)} 份，测距记录合计 {total_records} 条")
    if not args.apply:
        print("\n以上为预览。加 --apply 参数执行删除。")
        return 0
    ids = [row["id"] for row in rows]
    marks = ",".join("?" * len(ids))
    with db.transaction() as conn:
        conn.execute(f"DELETE FROM ranging_result WHERE experiment_id IN ({marks})", ids)
        conn.execute(f"DELETE FROM detection_result WHERE experiment_id IN ({marks})", ids)
        conn.execute(f"DELETE FROM segmentation_result WHERE experiment_id IN ({marks})", ids)
        conn.execute(f"DELETE FROM error_analysis WHERE experiment_id IN ({marks})", ids)
        conn.execute(f"DELETE FROM report WHERE experiment_id IN ({marks})", ids)
        conn.execute(f"DELETE FROM task_queue WHERE experiment_id IN ({marks})", ids)
        conn.execute(f"DELETE FROM experiment WHERE id IN ({marks})", ids)
    print(f"\n已删除 {len(ids)} 个演示实验及其产物。")
    print("剩余数据：")
    for table, count in db.table_counts().items():
        if count:
            print(f"  {table:<22} {count}")
    return 0
if __name__ == "__main__":
    raise SystemExit(main())

class Client:
    def __init__(self, base: str) -> None:
        self.base = base.rstrip("/")
        self.opener = urllib.request.build_opener(
            urllib.request.HTTPCookieProcessor(CookieJar()))
        self.token = ""
    def call(self, method: str, path: str, body: dict | None = None) -> dict:
        url = self.base + path
        data = json.dumps(body).encode() if body is not None else None
        request = urllib.request.Request(url, data=data, method=method)
        request.add_header("Content-Type", "application/json")
        if self.token:
            request.add_header("Authorization", "Bearer " + self.token)
        try:
            with self.opener.open(request, timeout=120) as response:
                payload = response.read()
                try:
                    return json.loads(payload)
                except json.JSONDecodeError:
                    return {"code": 0, "raw_length": len(payload)}
        except urllib.error.HTTPError as exc:
            text = exc.read().decode("utf-8", "ignore")
            try:
                return json.loads(text)
            except json.JSONDecodeError:
                return {"code": exc.code, "message": text[:200]}
    def raw(self, path: str) -> bytes:
        url = self.base + path
        request = urllib.request.Request(url)
        if self.token:
            request.add_header("Authorization", "Bearer " + self.token)
        with self.opener.open(request, timeout=120) as response:
            return response.read()

"""导出软件著作权登记所需的源代码文档（前 30 页 + 后 30 页，每页 50 行）。

]
SKIP_PARTS = (".venv", "venv", "__pycache__", "storage", "node_modules", ".git")
def collect() -> list[Path]:
    files: list[Path] = []
    for rel in FILE_ORDER:
        path = ROOT / rel
        if path.exists() and path.is_file():
            files.append(path)
    # 补充未被显式列出的源码文件
    for path in sorted(ROOT.rglob("*")):
        if not path.is_file():
            continue
        if any(part in path.parts for part in SKIP_PARTS):
            continue
        if path.suffix.lower() not in (".py", ".js", ".html", ".css", ".yaml"):
            continue
        if path.name.startswith("_"):
            continue
        if path not in files:
            files.append(path)
    return files
def build_lines(files: list[Path]) -> tuple[list[str], dict]:
    lines: list[str] = []
    per_file: dict[str, int] = {}
    for path in files:
        try:
            text = path.read_text(encoding="utf-8", errors="ignore")
        except OSError:
            continue
        content = [line.rstrip("\n") for line in text.splitlines()]
        # 去掉纯空行以符合「每页不少于 50 行」的排版要求
        content = [line for line in content if line.strip()]
        if not content:
            continue
        rel = path.relative_to(ROOT).as_posix()
        per_file[rel] = len(content)
        lines.append(f"// ==================== {rel} ====================")
        lines.extend(content)
        lines.append("")
    return lines, per_file
def main() -> int:
    parser = argparse.ArgumentParser(description="导出软著源代码文档")
    parser.add_argument("--pages", type=int, default=30, help="前后各取多少页（默认 30）")
    parser.add_argument("--lines", type=int, default=50, help="每页行数（默认 50）")
    args = parser.parse_args()
    files = collect()
    lines, per_file = build_lines(files)
    page_size = int(args.lines)
    needed = int(args.pages) * page_size
    out_dir = settings.storage_dir("exports")
    header = (f"视界智研算法实验平台 V1.0  源代码\n"
              f"导出时间：{datetime.now():%Y-%m-%d %H:%M:%S}\n"
              f"文件总数：{len(files)}   有效代码行：{len(lines)}\n")
    head_lines = lines[:needed]
    tail_lines = lines[-needed:] if len(lines) > needed else lines
    (out_dir / "软著源代码_前30页.txt").write_text(
        header + "\n".join(head_lines), encoding="utf-8")
    (out_dir / "软著源代码_后30页.txt").write_text(
        header + "\n".join(tail_lines), encoding="utf-8")
    stats = {
        "generated_at": datetime.now().isoformat(timespec="seconds"),
        "file_count": len(files),
        "code_lines": len(lines),
        "page_size": page_size,
        "pages_each_side": args.pages,
        "top_files": sorted(per_file.items(), key=lambda kv: -kv[1])[:20],
    }
    (out_dir / "软著源代码_统计.json").write_text(
        json.dumps(stats, ensure_ascii=False, indent=2), encoding="utf-8")
    print("=" * 68)
    print(f"源码文件 {len(files)} 个，有效代码行 {len(lines)} 行")
    print(f"前 {args.pages} 页 / 后 {args.pages} 页已导出至：{out_dir}")
    print("=" * 68)
    for rel, count in stats["top_files"][:10]:
        print(f"  {count:>6} 行  {rel}")
    return 0
if __name__ == "__main__":
    raise SystemExit(main())

// ==================== tools/tidy_state.py ====================
"""维护工具：把校正方案与中间分析记录收敛到最终交付状态。
只保留在真实历史数据上拟合的校正方案，并启用其中 RMSE 最低的一个。
用法：python tools/tidy_state.py
"""
from __future__ import annotations
import sys
from pathlib import Path
ROOT = Path(__file__).resolve().parent.parent
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))
from core import db  # noqa: E402
from core.logging_setup import setup_logging  # noqa: E402
KEEP_PREFIXES = ("历史数据", "车辆")
def main() -> int:
    setup_logging()
    db.init_db()
    patterns = " OR ".join(["name LIKE ?"] * len(KEEP_PREFIXES))
    params = [p + "%" for p in KEEP_PREFIXES]
    with db.transaction() as conn:
        conn.execute(f"DELETE FROM calibration_plan WHERE NOT ({patterns})", params)
        conn.execute("DELETE FROM calibration")
        conn.execute("DELETE FROM error_analysis")
        conn.execute("UPDATE calibration_plan SET is_active=0")
        # 平台默认启用「全量分段校正」；按类别的方案保留为可选，不默认启用
        row = conn.execute(
            "SELECT id FROM calibration_plan WHERE name='历史数据_piecewise' LIMIT 1").fetchone()
        if row is None:
            row = conn.execute(
                "SELECT id FROM calibration_plan WHERE method='piecewise' "
                "ORDER BY sample_count DESC LIMIT 1").fetchone()
        if row:
            conn.execute("UPDATE calibration_plan SET is_active=1 WHERE id=?", (row[0],))
    print("保留的校正方案：")
    for item in db.query_all("SELECT name, method, sample_count, rmse_before, rmse_after, "
                             "is_active FROM calibration_plan ORDER BY created_at"):
        flag = "已启用" if item["is_active"] else "      "
        print(f"  [{flag}] {item['name']:<20} {item['method']:<10} "
              f"样本 {item['sample_count']:<5} RMSE {item['rmse_before']} -> {item['rmse_after']}")
    return 0
if __name__ == "__main__":
    raise SystemExit(main())

// ==================== tools/tune_seg_threshold.py ====================
"""分割二值化阈值标定：为每个结构找到使 IoU 最优的阈值。
背景：原项目 B 的在线推理脚本用固定阈值 0.30，而各结构的概率分布不同，
最优阈值差异明显。本脚本在真实测试图上扫描阈值，为平台默认值与界面滑块提供依据。
用法：python tools/tune_seg_threshold.py [数据根目录] [权重目录]
"""
from __future__ import annotations
import sys
from pathlib import Path
ROOT = Path(__file__).resolve().parent.parent
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))
import cv2  # noqa: E402
import numpy as np  # noqa: E402
from algorithms import registry  # noqa: E402
from core import media  # noqa: E402
from core.logging_setup import setup_logging  # noqa: E402
DEFAULT_DATA = Path(r"E:\2Project\2026dachuang\Unet")
PLAN = [("unet", "Unet_1000_best.pth"), ("resunet", "resunet_1000_best.pth"),
        ("att_resunet", "Atteresunet_1000_best.pth"), ("unetv2", "UNetV2_800_best.pth")]
THRESHOLDS = [0.05, 0.10, 0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90]
REPORTED = {"unet": 0.9688, "resunet": 0.9777, "att_resunet": 0.9737, "unetv2": 0.9364}
def iou_of(pred: np.ndarray, gt: np.ndarray) -> float:
    p = (pred > 0).ravel()
    g = (gt > 0).ravel()
    tp = float(np.logical_and(p, g).sum())
    fp = float(np.logical_and(p, ~g).sum())
    fn = float(np.logical_and(~p, g).sum())
    return tp / (tp + fp + fn + 1e-7)
def main() -> int:
    data_root = Path(sys.argv[1]) if len(sys.argv) > 1 else DEFAULT_DATA
    weights_dir = Path(sys.argv[2]) if len(sys.argv) > 2 else data_root / "models"
    setup_logging()
    registry.load_builtin()
    test_images = sorted((data_root / "dataset" / "test" / "images").glob("*.jpg"))[:6]
    masks_dir = data_root / "dataset" / "test" / "masks"
    if not test_images:
        print("找不到测试图像。")
        return 1
    print(f"抽检 {len(test_images)} 张测试图\n")
    verdicts = {}
    for arch, weight in PLAN:
        plugin = registry.create("segmentation", arch)
        plugin.load(str(weights_dir / weight))
        scores = []
        for threshold in THRESHOLDS:
            ious = []
            for image_path in test_images:
                image = media.read_image(image_path)
                raw = cv2.imread(str(masks_dir / (image_path.stem + ".png")), cv2.IMREAD_UNCHANGED)
                if raw is None:
                    continue
                if raw.ndim == 3:
                    raw = raw[:, :, 0]
                pred = plugin.predict(image, threshold=threshold)["mask"]
                ious.append(iou_of(pred, (raw > 0).astype(np.uint8)))
            scores.append((threshold, float(np.mean(ious))))
        best = max(scores, key=lambda kv: kv[1])
        reference = REPORTED.get(arch, 0.97)
        closest = min(scores, key=lambda kv: abs(kv[1] - reference))
        verdicts[arch] = {"best_threshold": best[0], "best_iou": round(best[1], 4),
                          "closest_threshold": closest[0], "iou_at_closest": round(closest[1], 4),
                          "reported_iou": reference}
        print(f"== {arch} ==")
        print("   " + "  ".join(f"t={t:.2f}:{v:.4f}" for t, v in scores))
        print(f"   最佳阈值 {best[0]:.2f} -> IoU {best[1]:.4f}；"
              f"最接近原项目 B 报告值({reference})的阈值 {closest[0]:.2f} -> IoU {closest[1]:.4f}\n")
    print("=" * 84)
    print("结论：各结构最优阈值并不相同，平台界面提供阈值滑块以便按结构调优")
    print("=" * 84)
    for arch, value in verdicts.items():
        print(f"  {arch:<12} 原报告 IoU={value['reported_iou']}  "
              f"最接近阈值={value['closest_threshold']}  实测 IoU={value['iou_at_closest']}  "
              f"(最优 {value['best_iou']} @ t={value['best_threshold']})")
    return 0
if __name__ == "__main__":
    raise SystemExit(main())
